## composite build / buildSrc を利用している場合

`gradle()` は `build-logic` などの included build と `buildSrc` を宣言しません。その中のビルドスクリプトと規約プラグインは、自分で宣言します。

```kt title="ProjectArchitecture.kt"
import me.tbsten.katachi.dsl.architecture
import me.tbsten.katachi.dsl.gradle.gradle
import me.tbsten.katachi.dsl.kotlin.ktFile
import me.tbsten.katachi.dsl.kotlin.ktsFile

const val buildConventionDir = "build-logic" // buildSrc を利用する場合は代わりに "buildSrc"

val architecture = architecture {
    gradle()

    "Build".group {
        documented = false

        "BuildScript" {
            layout {
                buildConventionDir {
                    "build.gradle".ktsFile()
                    "settings.gradle".ktsFile().optional() // buildSrc では無くてもよい
                }
            }
        }

        "ConventionPlugins" {
            layout {
                "$buildConventionDir/src/main/kotlin" {
                    "*.gradle".ktsFile()
                    "**" / "*".ktFile() // パッケージを切ったクラスも許す
                }
            }

            // ./gradlew katachiTemplate --arg roleName=Build/ConventionPlugins --arg name=kotlin-library
            template {
                val name by stringParameter()

                file("${name}.gradle.kts") {
                    """
                        plugins {
                            kotlin("jvm")
                        }
                    """.trimIndent()
                }
            }
        }
    }
}
```
