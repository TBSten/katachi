// ProjectArchitecture.kt
@file:OptIn(ExperimentalKatachiApi::class)

import me.tbsten.katachi.ExperimentalKatachiApi
import me.tbsten.katachi.dsl.architecture

val projectArchitecture = architecture {
    "AiAgent".group {
        claudeCode()
    }
}

// roles/ClaudeCode.kt
@file:OptIn(ExperimentalKatachiApi::class)

import me.tbsten.katachi.ExperimentalKatachiApi
import me.tbsten.katachi.dsl.DeclarationContainerScope

fun DeclarationContainerScope.claudeCode() =
    "ClaudeCode".group {
        "Rules" {
            summary = "Claude Code に読ませる指示ファイル"

            layout {
                "**/CLAUDE.md".file()
                "**/AGENTS.md".file()
                ".claude/rules/**/*.md".file()
            }
        }

        "Skills" {
            summary = "Claude Code のスキル"

            layout {
                ".claude/skills/**/*".file()
            }
        }

        "Agents" {
            summary = "Claude Code のサブエージェント定義"

            layout {
                ".claude/agents/**/*.md".file()
            }
        }

        "Commands" {
            summary = "Claude Code のスラッシュコマンド"

            layout {
                ".claude/commands/**/*.md".file()
            }
        }

        "OutputStyles" {
            summary = "Claude Code の出力スタイル"

            layout {
                ".claude/output-styles/*.md".file()
            }
        }

        "Hooks" {
            summary = "Claude Code のフックスクリプト"

            layout {
                ".claude/hooks/**/*".file()
            }
        }

        "Settings" {
            summary = "Claude Code の設定ファイル"

            layout {
                ".claude/settings.json".file().optional()
            }
        }

        "Mcp" {
            summary = "Claude Code の MCP サーバー設定"

            layout {
                ".mcp.json".file().optional()
            }
        }
    }
