You can declare the configuration files of nine AI coding tools, such as Claude Code and Cursor, with one group and Roles per tool.

## Claude Code

Call `claudeCode()` from `ProjectArchitecture.kt`, and put its body in `roles/ClaudeCode.kt`.

```kt title=ProjectArchitecture.kt nocollapse
@file:OptIn(ExperimentalKatachiApi::class)

import me.tbsten.katachi.ExperimentalKatachiApi
import me.tbsten.katachi.dsl.architecture

val projectArchitecture = architecture {
  "AiAgent".group {
    claudeCode()
  }
}
```

```kt title=roles/ClaudeCode.kt
@file:OptIn(ExperimentalKatachiApi::class)

import me.tbsten.katachi.ExperimentalKatachiApi
import me.tbsten.katachi.dsl.DeclarationContainerScope

fun DeclarationContainerScope.claudeCode() =
  "ClaudeCode".group {
    documented = false

    "Rules".role {
      summary = "Instruction files that Claude Code reads"

      layout {
        "**/CLAUDE.md".file()
        "**/AGENTS.md".file()
        ".claude/rules/**/*.md".file()
      }
    }

    "Skills".role {
      summary = "Claude Code skills"

      layout {
        ".claude/skills/**/*".file()
      }
    }

    "Agents".role {
      summary = "Claude Code subagent definitions"

      layout {
        ".claude/agents/**/*.md".file()
      }
    }

    "Commands".role {
      summary = "Claude Code slash commands"

      layout {
        ".claude/commands/**/*.md".file()
      }
    }

    "OutputStyles".role {
      summary = "Claude Code output styles"

      layout {
        ".claude/output-styles/*.md".file()
      }
    }

    "Hooks".role {
      summary = "Claude Code hook scripts"

      layout {
        ".claude/hooks/**/*".file()
      }
    }

    "Settings".role {
      summary = "Claude Code settings file"

      layout {
        ".claude/settings.json".file().optional()
      }
    }

    "Mcp".role {
      summary = "Claude Code MCP server configuration"

      layout {
        ".mcp.json".file().optional()
      }
    }
  }
```
