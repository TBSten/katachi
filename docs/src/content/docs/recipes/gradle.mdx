## When you use a composite build / buildSrc

`gradle()` does not declare included builds such as `build-logic`, nor `buildSrc`. Declare their build scripts and convention plugins yourself.

```kt title="ProjectArchitecture.kt"
import me.tbsten.katachi.dsl.architecture
import me.tbsten.katachi.dsl.gradle.gradle
import me.tbsten.katachi.dsl.kotlin.ktFile
import me.tbsten.katachi.dsl.kotlin.ktsFile

const val buildConventionDir = "build-logic" // Use "buildSrc" instead if you use buildSrc

val architecture = architecture {
    gradle()

    "Build".group {
        documented = false

        "BuildScript" {
            layout {
                buildConventionDir {
                    "build.gradle".ktsFile()
                    "settings.gradle".ktsFile().optional() // buildSrc may go without one
                }
            }
        }

        "ConventionPlugins" {
            layout {
                "$buildConventionDir/src/main/kotlin" {
                    "*.gradle".ktsFile()
                    "**" / "*".ktFile() // also allows classes in packages
                }
            }

            // ./gradlew katachiTemplate --arg roleName=Build/ConventionPlugins --arg name=kotlin-library
            template {
                val name by stringParameter()

                file("${name}.gradle.kts") {
                    """
                        plugins {
                            kotlin("jvm")
                        }
                    """.trimIndent()
                }
            }
        }
    }
}
```
