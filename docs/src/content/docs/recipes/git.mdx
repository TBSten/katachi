{/*
  The purpose of this page: to show, in a copy-and-paste form, how to declare .gitignore / .gitkeep / .gitattributes as one Role per kind of file.
*/}

import SnippetIsStartingPoint from '../../../components/SnippetIsStartingPoint.astro';

```kt collapse {6,15,23,31}
// ProjectArchitecture.kt
import me.tbsten.katachi.dsl.architecture

val projectArchitecture = architecture {
    "Tools".group {
        git()
    }
}

// roles/Git.kt
import me.tbsten.katachi.dsl.DeclarationContainerScope

fun DeclarationContainerScope.git() =
    "Git".group {
        "GitIgnore" {
            title = "Git ignore"
            summary = "The list of files kept out of Git (.gitignore)"
            layout {
                "**/.gitignore".file()
            }
        }

        "GitKeep" {
            title = "Git keep"
            summary = "Empty placeholder files that keep a directory in Git (.gitkeep)"
            layout {
                "**/.gitkeep".file()
            }
        }

        "GitAttributes" {
            title = "Git attributes"
            summary = "Per-path attributes such as line endings and diff handling (.gitattributes)"
            layout {
                "**/.gitattributes".file()
            }
        }
    }
```
